import time
import sys

def type_animate(text, delay=0.1):
    """Function to animate typing each character"""
    for char in text:
        sys.stdout.write(char)
        sys.stdout.flush()
        time.sleep(delay)
    print()

def generate_post_chorus():
    post_chorus_lines = [
        "'Di mo ba ako lilisanin?",

        "Hindi na ba ma-mamamayapa?",

    ]

    # Delays (to match song timing)
    delays = [1.5, 0.1, 1.5, 1.5, 2.5]

    for line, delay in zip(post_chorus_lines, delays):
        type_animate(line)
        time.sleep(delay)

import pygame
import random

# Initialize Pygame
pygame.init()

# Screen settings
WIDTH, HEIGHT = 600, 400
screen = pygame.display.set_mode((WIDTH, HEIGHT))
pygame.display.set_caption("Catch the Ball")

# Colors
WHITE = (255, 255, 255)
BLUE = (50, 50, 255)
RED = (255, 50, 50)
BLACK = (0, 0, 0)

# Game variables
player_width, player_height = 100, 20
player_x = (WIDTH - player_width) // 2
player_y = HEIGHT - player_height - 10
player_speed = 7

ball_radius = 15
ball_x = random.randint(ball_radius, WIDTH - ball_radius)
ball_y = -ball_radius
ball_speed = 5

score = 0
font = pygame.font.SysFont(None, 36)

clock = pygame.time.Clock()
running = True

while running:
    screen.fill(WHITE)

    # Event handling
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

    # Player movement
    keys = pygame.key.get_pressed()
    if keys[pygame.K_LEFT] and player_x > 0:
        player_x -= player_speed
    if keys[pygame.K_RIGHT] and player_x < WIDTH - player_width:
        player_x += player_speed

    # Ball movement
    ball_y += ball_speed

    # Catching the ball
    if (player_y < ball_y + ball_radius < player_y + player_height) and \
       (player_x < ball_x < player_x + player_width):
        score += 1
        ball_y = -ball_radius
        ball_x = random.randint(ball_radius, WIDTH - ball_radius)

    # Missed ball
    if ball_y > HEIGHT:
        ball_y = -ball_radius
        ball_x = random.randint(ball_radius, WIDTH - ball_radius)

    # Drawing
    pygame.draw.rect(screen, BLUE, (player_x, player_y, player_width, player_height))
    pygame.draw.circle(screen, RED, (ball_x, ball_y), ball_radius)
    score_text = font.render(f"Score: {score}", True, BLACK)
    screen.blit(score_text, (10, 10))

    pygame.display.flip()
    clock.tick(60)

pygame.quit()
